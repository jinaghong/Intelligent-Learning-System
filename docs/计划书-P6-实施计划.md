> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 tlias 后端新增 8 个只读统计接口（班级人数 / 学历 / 性别 / 课程成绩 / 分数段 / 班级违纪 / 违纪 Top10 / 部门人数），配 20 条验收用例与独立对账。

**Tech Stack:** JDK 21 / Spring Boot 3.5.16 / MyBatis-Plus 3.5.17 / MySQL 8.0.36 / JUnit 5 + MockMvc + JdbcTemplate / Lombok。

```java
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.hjh.tlias.config.properties.JwtProperties;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.DisplayName;

 * P6 数据统计的验收测试（先写、先跑红，再实现）。

 * 分三批与实现交替推进：
 *   批次一：班级人数 / 学历 / 性别（本批）

 * 关键断言用 JdbcTemplate 写「独立 SQL」对账——防止实现与断言同源同错。

@SpringBootTest
@AutoConfigureMockMvc
@Transactional
class ReportFlowTest {

    private static final String PASSWORD = "123456";

    private MockMvc mockMvc;

    @Autowired

    @Autowired

    @Test
    @DisplayName("7. 性别分布：2 行、男+女=48、名字正确")
    void studentGenderTwoRows() throws Exception {
        JsonNode rows = getData("/report/student-gender");

        System.out.println("\n[7] 性别分布 = " + rows);
        assertEquals(2, rows.size());
        assertEquals(1, rows.get(0).path("gender").asInt());
        assertEquals("男", rows.get(0).path("genderName").asText());
        assertEquals(2, rows.get(1).path("gender").asInt());

        assertEquals(48, rows.get(0).path("count").asInt() + rows.get(1).path("count").asInt());

    @Test
    @DisplayName("8. 未登录访问统计接口 → 401")
    void reportRequiresAuth() throws Exception {
        mockMvc.perform(get("/report/clazz-student-count")).andExpect(status().isUnauthorized());
    }

    // ------------------------------------------------------------------
    //  测试自己的小工具
    // ------------------------------------------------------------------

    /** 发 GET 并返回 data 节点（要求 code=1） */
    private JsonNode getData(String uri) throws Exception {
        String body = mockMvc.perform(get(uri).header(jwtProperties.getTokenName(), token))
                .andExpect(status().isOk()).andExpect(jsonPath("$.code").value(1))
                .andReturn().getResponse().getContentAsString(StandardCharsets.UTF_8);
        return objectMapper.readTree(body).path("data");
    }

    /** 在名单里按名字字段找一行；找不到直接失败（比空指针好排查） */
    private JsonNode findRow(JsonNode rows, String nameField, String name) {
        for (JsonNode row : rows) {
            if (name.equals(row.path(nameField).asText())) {
                return row;
            }
        }
        throw new AssertionError("结果里找不到 " + nameField + " = " + name);
    }

    private int totalOf(JsonNode rows, String clazzName) {
        return findRow(rows, "clazzName", clazzName).path("total").asInt();
    }

}
```

```java
package com.hjh.tlias.mapper;

import com.hjh.tlias.vo.ClazzStudentCountVO;
import com.hjh.tlias.vo.StudentDegreeVO;
import com.hjh.tlias.vo.StudentGenderVO;

import java.util.List;

/**
 * 数据统计 Mapper。
 *
 * 不继承 BaseMapper：统计查询没有对应的「单表实体」，
 * 每个方法都是一条聚合 SQL，全部写在 ReportMapper.xml（P4 惯例：复杂 SQL 走 XML）。
 *
 * 不需要 @Mapper 注解：主启动类上的 @MapperScan("com.hjh.tlias.mapper") 统一扫描。
 */
public interface ReportMapper {

    List<ClazzStudentCountVO> clazzStudentCount();

    List<StudentDegreeVO> studentDegree();

    List<StudentGenderVO> studentGender();

}
```

`resources/mapper/ReportMapper.xml`：

```xml
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE mapper PUBLIC "-//mybatis.org//DTD Mapper 3.0//EN"
        "http://mybatis.org/dtd/mybatis-3-mapper.dtd">

<mapper namespace="com.hjh.tlias.mapper.ReportMapper">

    <!--
      班级人数：以 clazz 为基准 LEFT JOIN，空班级以 0 出现（P4 老教训：不许丢行）。
      SUM(条件) 是 MySQL 的写法：条件为真算 1、为假算 0，正好当计数器用；
      空班级时 SUM 是 NULL，用 COALESCE 兜 0。
    -->
    <select id="clazzStudentCount" resultType="com.hjh.tlias.vo.ClazzStudentCountVO">
        SELECT c.name                          AS clazzName,
               COUNT(s.id)                     AS total,
               COALESCE(SUM(s.gender = 1), 0)  AS maleCount,
               COALESCE(SUM(s.gender = 2), 0)  AS femaleCount
        FROM clazz c
                 LEFT JOIN student s ON s.clazz_id = c.id
        GROUP BY c.id
        ORDER BY c.id
    </select>

    <!-- 学历分布：只出「编码 + 数量」，中文名与缺失补 0 由 Service 层做 -->
    <select id="studentDegree" resultType="com.hjh.tlias.vo.StudentDegreeVO">
        SELECT degree, COUNT(*) AS count
        FROM student
        GROUP BY degree
        ORDER BY degree
    </select>

    <!-- 性别分布：同上 -->
    <select id="studentGender" resultType="com.hjh.tlias.vo.StudentGenderVO">
        SELECT gender, COUNT(*) AS count
        FROM student
        GROUP BY gender
        ORDER BY gender
    </select>

</mapper>
```

```java
import com.hjh.tlias.service.ReportService;
import com.hjh.tlias.vo.ClazzStudentCountVO;

import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;

/**
 * 数据统计实现。
 *
 * 分工：Mapper 只负责「编码 + 数量」的纯聚合；
 * 中文标签与「缺失类目补 0」在这里做——保证封闭枚举（学历 6 类 / 性别 2 类）
 * 行数恒定，P9 的图坐标轴不随数据抖动。
 */
@Service
@RequiredArgsConstructor
public class ReportServiceImpl implements ReportService {

    /** 下标 = 学历编码；0 号位空占位，让 degree 直接当下标用 */
    private static final String[] DEGREE_NAMES = {"", "初中", "高中", "专科", "本科", "硕士", "博士"};

    private final ReportMapper reportMapper;

    @Override
    public List<ClazzStudentCountVO> clazzStudentCount() {
        return reportMapper.clazzStudentCount();
    }
```

```xml
      课程成绩：以 course 为基准 LEFT JOIN，无成绩的课程 COUNT=0、
      AVG/MAX/MIN 为 NULL（行保留，前端自行决定怎么展示）。
      及格率 = 及格人次(分数 >= 60) / 考试人次 × 100，保留 1 位小数；
      注意：SQL 里的 >= 在 XML 中必须转义成 &gt;= 。
    -->
    <select id="courseScore" resultType="com.hjh.tlias.vo.CourseScoreVO">
        SELECT co.name                                        AS courseName,
               COUNT(ss.id)                                   AS examCount,
               ROUND(AVG(ss.score), 2)                        AS avgScore,
               MAX(ss.score)                                  AS maxScore,
               MIN(ss.score)                                  AS minScore,
               ROUND(SUM(ss.score &gt;= 60) * 100.0 / COUNT(ss.id), 1) AS passRate
        FROM course co
                 LEFT JOIN student_score ss ON ss.course_id = co.id
        GROUP BY co.id
        ORDER BY co.id
    </select>

                   WHEN score &gt;= 90 THEN '90-100'
                   WHEN score &gt;= 80 THEN '80-89'
                   WHEN score &gt;= 70 THEN '70-79'
                   WHEN score &gt;= 60 THEN '60-69'
                   ELSE '0-59'
                   END  AS `range`,
               COUNT(*) AS count
        FROM student_score
        GROUP BY `range`
    </select>
```

`ReportMapper.xml` 在 `</mapper>` 前追加：

```xml
    <!-- 班级违纪：双 LEFT JOIN 链；无违纪班级 COUNT=0、SUM 为 NULL → COALESCE 兜 0 -->
    <select id="clazzViolation" resultType="com.hjh.tlias.vo.ClazzViolationVO">
        SELECT c.name                    AS clazzName,
               COUNT(v.id)               AS violationCount,
               COALESCE(SUM(v.point), 0) AS totalPoints
        FROM clazz c
                 LEFT JOIN student s ON s.clazz_id = c.id
                 LEFT JOIN student_violation v ON v.student_id = s.id
        GROUP BY c.id
        ORDER BY c.id
    </select>

    <!--
      违纪 Top10：直接用 student 的汇总字段，稳定排序三键
      （扣分降序 → 次数降序 → id 升序），并列时不抖动。
    -->
    <select id="violationTop" resultType="com.hjh.tlias.vo.ViolationTopVO">
        SELECT s.name            AS studentName,
               c.name            AS clazzName,
               s.violation_count AS violationCount,
               s.violation_score AS violationScore
        FROM student s
                 LEFT JOIN clazz c ON s.clazz_id = c.id
        WHERE s.violation_count &gt; 0
        ORDER BY s.violation_score DESC, s.violation_count DESC, s.id ASC
        LIMIT 10
    </select>

    <!-- 部门人数：以 dept 为基准；0 人部门以 0 出现 -->
    <select id="deptEmpCount" resultType="com.hjh.tlias.vo.DeptEmpCountVO">
        SELECT d.name                          AS deptName,
               COUNT(e.id)                     AS total,
               COALESCE(SUM(e.status = 1), 0)  AS activeCount,
               COALESCE(SUM(e.status = 2), 0)  AS inactiveCount
        FROM dept d
                 LEFT JOIN emp e ON e.dept_id = d.id
        GROUP BY d.id
        ORDER BY d.id
    </select>
```
